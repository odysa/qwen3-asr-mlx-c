cmake_minimum_required(VERSION 3.20)
project(qwen3_asr LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_CXX_STANDARD 20)
set(MLX_C_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
add_subdirectory(deps/mlx-c EXCLUDE_FROM_ALL)
add_library(qwen_weights src/weights.c)
target_include_directories(qwen_weights PUBLIC include deps/mlx-c)
target_compile_definitions(qwen_weights PUBLIC QWEN_WITH_MLX_C=1)
target_link_libraries(qwen_weights PUBLIC mlxc)
set(QWEN_MODEL_SOURCES src/tokenizer.c)
foreach(_src IN ITEMS src/model.c src/kernels.c src/audio.c)
  if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/${_src}")
    list(APPEND QWEN_MODEL_SOURCES "${_src}")
  endif()
endforeach()
add_executable(qwen3-asr src/main.c ${QWEN_MODEL_SOURCES})
target_link_libraries(qwen3-asr PRIVATE qwen_weights)
target_include_directories(qwen3-asr PRIVATE include deps/mlx-c)
add_executable(qwen3-asr-resident-bench tools/resident_bench.c ${QWEN_MODEL_SOURCES})
target_link_libraries(qwen3-asr-resident-bench PRIVATE qwen_weights)
target_include_directories(qwen3-asr-resident-bench PRIVATE include deps/mlx-c)

add_executable(qwen3-asr-worker src/serve_worker.c ${QWEN_MODEL_SOURCES})
target_link_libraries(qwen3-asr-worker PRIVATE qwen_weights)
target_include_directories(qwen3-asr-worker PRIVATE include deps/mlx-c)

option(QWEN_EXPERIMENTS "Build experimental quantized model variants" OFF)
if(QWEN_EXPERIMENTS)
  foreach(variant IN ITEMS q8 q4 head8)
    add_executable(qwen3-asr-${variant}-worker src/serve_worker.c experimental/model_${variant}.c src/tokenizer.c src/audio.c)
    target_link_libraries(qwen3-asr-${variant}-worker PRIVATE qwen_weights)
    target_include_directories(qwen3-asr-${variant}-worker PRIVATE include deps/mlx-c)
    add_executable(qwen3-asr-${variant} src/main.c experimental/model_${variant}.c src/tokenizer.c src/audio.c)
    target_link_libraries(qwen3-asr-${variant} PRIVATE qwen_weights)
    target_include_directories(qwen3-asr-${variant} PRIVATE include deps/mlx-c)
    add_executable(qwen3-asr-${variant}-resident experimental/resident_memory_bench.c experimental/model_${variant}.c src/tokenizer.c src/audio.c)
    target_link_libraries(qwen3-asr-${variant}-resident PRIVATE qwen_weights)
    target_include_directories(qwen3-asr-${variant}-resident PRIVATE include deps/mlx-c)
  endforeach()
endif()
